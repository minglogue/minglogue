/****************************************************
 * 설치형 onEdit 트리거용 메인 함수
 ****************************************************/

function installedOnEdit(e) {
  if (!e || !e.range) return;

  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== CONFIG.mainSheetName) return;
  if (range.getLastRow() < CONFIG.startRow) return;

  const startCol = range.getColumn();
  const endCol = range.getLastColumn();

  const touched = function(column) {
    return startCol <= column && endCol >= column;
  };

  // F열은 가장 먼저 처리하고 즉시 종료한다.
  if (touched(CONFIG.col.doctorReady)) {
    safeRunOnEditHandler_('onReadyJandi_', function() {
      onReadyJandi_(e);
    });



    return;
  }

  if (touched(CONFIG.col.customer)) {
    safeRunOnEditHandler_('handleReceptionTimestampToAC_', function() {
      handleReceptionTimestampToAC_(e);
    });
    safeRunOnEditHandler_('handleCheckboxTimestampToAB_', function() {
      handleCheckboxTimestampToAB_(e);
    });
    safeRunOnEditHandler_('drawCustomerGroupLines_', function() {
      drawCustomerGroupLines_(e);
    });
  }

  if (touched(CONFIG.col.doctor)) {
    safeRunOnEditHandler_('handleDoctorTimestampToAD_', function() {
      handleDoctorTimestampToAD_(e);
    });
  }

  if (touched(CONFIG.col.skin1)) {
    safeRunOnEditHandler_('handleSkin1TimestampToAE_', function() {
      handleSkin1TimestampToAE_(e);
    });
  }

  if (
    startCol <= CONFIG.col.anesthesiaStart &&
    endCol >= CONFIG.col.anesthesiaDuration
  ) {
    safeRunOnEditHandler_('handleAnesthesiaEndTime_', function() {
      handleAnesthesiaEndTime_(e);
    });
  }
}


/****************************************************
 * 잔디 알림
 ****************************************************/

function onReadyJandi_(e) {
  if (!e || !e.range || !CONFIG.jandiWebhookUrl) return;

  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== CONFIG.mainSheetName) return;
  if (!rangeTouchesColumn_(range, CONFIG.col.doctorReady)) return;

  const rowStart = Math.max(range.getRow(), CONFIG.startRow);
  const rowEnd = range.getLastRow();
  const numRows = rowEnd - rowStart + 1;

  if (numRows <= 0) return;

  const readyValues = sheet
    .getRange(rowStart, CONFIG.col.doctorReady, numRows, 1)
    .getValues();

  for (let i = 0; i < numRows; i++) {
    const row = rowStart + i;
    const checked = isTrue_(readyValues[i][0]);
    const notificationKey = updateReadyNotificationState_(
      sheet,
      row,
      checked
    );

    if (!notificationKey) continue;
    sendReadyJandiForRow_(sheet, row, notificationKey);
  }
}

function updateReadyNotificationState_(sheet, row, checked) {
  const lock = LockService.getScriptLock();

  if (!lock.tryLock(5000)) {
    console.log('JANDI_SKIP: state lock row=' + row);
    return null;
  }

  try {
    const properties = PropertiesService.getScriptProperties();
    const notificationKey =
      'jandi-ready-' + sheet.getSheetId() + '-' + row;

    if (!checked) {
      properties.deleteProperty(notificationKey);
      return null;
    }

    if (properties.getProperty(notificationKey)) {
      console.log('JANDI_SKIP: duplicate row=' + row);
      return null;
    }

    properties.setProperty(
      notificationKey,
      new Date().toISOString()
    );

    return notificationKey;
  } finally {
    lock.releaseLock();
  }
}

function sendReadyJandiForRow_(sheet, row, notificationKey) {
  const firstCol = CONFIG.col.customer;
  const lastCol = CONFIG.col.assignedDoctor;
  const values = sheet
    .getRange(row, firstCol, 1, lastCol - firstCol + 1)
    .getDisplayValues()[0];

  const valueAt = function(column) {
    return String(values[column - firstCol] || '').trim();
  };

  const customerName =
    valueAt(CONFIG.col.customer) || '고객명 없음';
  const treatmentName =
    valueAt(CONFIG.col.treatment) || '시술명 없음';
  const roomName =
    valueAt(CONFIG.col.room) || '시술실 미지정';
  const doctorName = valueAt(CONFIG.col.assignedDoctor);
  const doctorText = doctorName
    ? '지정원장 : **★★★ [' + doctorName + '] 원장님**'
    : '지정없음';

  const payload = {
    body:
      '[시술준비 완료]\n' +
      '고객명 : ' + customerName + '\n' +
      '시술명 : ' + treatmentName + '\n' +
      '시술실 : ' + roomName + '\n' +
      doctorText,
    connectColor: '#FAC11B'
  };

  const startedAt = Date.now();
  try {
    const response = UrlFetchApp.fetch(CONFIG.jandiWebhookUrl, {
      method: 'post',
      contentType: 'application/json',
      headers: {
        Accept: 'application/vnd.tosslab.jandi-v2+json'
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    const status = response.getResponseCode();

    if (status < 200 || status >= 300) {
      PropertiesService
        .getScriptProperties()
        .deleteProperty(notificationKey);
    }

    console.log(JSON.stringify({
      type: 'JANDI_SENT',
      row: row,
      status: status,
      elapsedMs: Date.now() - startedAt
    }));
  } catch (err) {
    PropertiesService
      .getScriptProperties()
      .deleteProperty(notificationKey);
    throw err;
  }
}

function handleCheckboxTimestampToAB_(e) {
  if (!e || !e.range) return;

  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== CONFIG.mainSheetName) return;
  if (!rangeTouchesColumn_(range, CONFIG.col.doctorReady)) return;

  const rowStart = Math.max(range.getRow(), CONFIG.startRow);
  const rowEnd = range.getRow() + range.getNumRows() - 1;
  const numRows = rowEnd - rowStart + 1;

  if (numRows <= 0) return;

  const readyValues = sheet
    .getRange(rowStart, CONFIG.col.doctorReady, numRows, 1)
    .getValues();

  const timeRange = sheet
    .getRange(rowStart, CONFIG.col.doctorCallTime, numRows, 1);

  const oldTimes = timeRange.getValues();
  const nowTime = getCurrentTimeOnlyValue_();

  const output = readyValues.map(function(row, index) {
    const checked = isTrue_(row[0]);
    const oldTime = oldTimes[index][0];

    if (!checked) {
      return [''];
    }

    if (!isBlank_(oldTime)) {
      return [normalizeToTimeOnlyValue_(oldTime)];
    }

    return [nowTime];
  });

  timeRange.setNumberFormat('HH:mm');
  timeRange.setValues(output);

}

function handleCheckboxTimestamp_(e, watchCol, timestampCol, keepExisting) {
  if (!e || !e.range) return;

  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== CONFIG.mainSheetName) return;
  if (!rangeTouchesColumn_(range, watchCol)) return;

  const rowStart = Math.max(range.getRow(), CONFIG.startRow);
  const rowEnd = range.getRow() + range.getNumRows() - 1;
  const numRows = rowEnd - rowStart + 1;

  if (numRows <= 0) return;

  const values = sheet.getRange(rowStart, watchCol, numRows, 1).getValues();
  const existing = sheet.getRange(rowStart, timestampCol, numRows, 1).getValues();
  const nowTime = getCurrentTimeOnlyValue_();

  const output = values.map(function(row, index) {
    const value = row[0];
    const oldTime = existing[index][0];

    if (!isTrue_(value)) return [''];
    if (keepExisting && !isBlank_(oldTime)) return [normalizeToTimeOnlyValue_(oldTime)];

    return [nowTime];
  });

  const targetRange = sheet.getRange(rowStart, timestampCol, numRows, 1);
  targetRange.setNumberFormat('HH:mm');
  targetRange.setValues(output);
}

/****************************************************
 * 마취/팩 종료시간 계산
 ****************************************************/

function handleAnesthesiaEndTime_(e) {
  if (!e || !e.range) return;

  const range = e.range;
  const sheet = range.getSheet();

  if (sheet.getName() !== CONFIG.mainSheetName) return;

  const rowStart = Math.max(range.getRow(), CONFIG.startRow);
  const rowEnd = range.getRow() + range.getNumRows() - 1;
  const numRows = rowEnd - rowStart + 1;

  if (numRows <= 0) return;

  const editedDuration =
    range.getColumn() <= CONFIG.col.anesthesiaDuration &&
    range.getLastColumn() >= CONFIG.col.anesthesiaDuration;

  calculateAnesthesiaEndTime_(sheet, rowStart, numRows, editedDuration);

}

function calculateAnesthesiaEndTime_(sheet, rowStart, numRows, shouldAutoFillStartTime) {
  if (numRows <= 0) return;

  const durationRawValues = sheet
  .getRange(rowStart, CONFIG.col.anesthesiaDuration, numRows, 1)
  .getValues();

  const durationDisplayValues = sheet
  .getRange(rowStart, CONFIG.col.anesthesiaDuration, numRows, 1)
  .getDisplayValues();

  const startRange = sheet.getRange(rowStart, CONFIG.col.anesthesiaStart, numRows, 1);
  const startValues = startRange.getValues();
  const startDisplayValues = startRange.getDisplayValues();

  const endRange = sheet.getRange(rowStart, CONFIG.col.anesthesiaEnd, numRows, 1);
  const endDisplayValues = endRange.getDisplayValues();

  const now = new Date();

  for (let i = 0; i < numRows; i++) {
    const currentRow = rowStart + i;
    const durationText = String(
      durationDisplayValues[i][0] || durationRawValues[i][0] || ''
    ).trim();

    let startValue = startValues[i][0];
    let startDisplayValue = String(startDisplayValues[i][0] || '').trim();
    const endDisplayValue = String(endDisplayValues[i][0] || '').trim();

    if (!durationText) {
      if (endDisplayValue !== '') {
        sheet.getRange(currentRow, CONFIG.col.anesthesiaEnd).clearContent();
      }
      continue;
    }

    const minuteMatch = durationText.match(/\d+/);
    if (!minuteMatch) {
      if (endDisplayValue !== '') {
        sheet.getRange(currentRow, CONFIG.col.anesthesiaEnd).clearContent();
      }
      continue;
    }

    const minutes = Number(minuteMatch[0]);
    if (!Number.isFinite(minutes) || minutes <= 0) {
      if (endDisplayValue !== '') {
        sheet.getRange(currentRow, CONFIG.col.anesthesiaEnd).clearContent();
      }
      continue;
    }

    const hasStartTime = !isBlank_(startValue) || startDisplayValue !== '';

    if (shouldAutoFillStartTime && !hasStartTime) {
      startValue = new Date(now);
      startDisplayValue = Utilities.formatDate(startValue, Session.getScriptTimeZone(), 'HH:mm');

      const startCell = sheet.getRange(currentRow, CONFIG.col.anesthesiaStart);
      startCell.setValue(startValue);
      startCell.setNumberFormat('HH:mm');
    }

    const startDate = parseTimeValue_(startValue, startDisplayValue);

    if (!startDate || isNaN(startDate.getTime())) {
      if (endDisplayValue !== '') {
        sheet.getRange(currentRow, CONFIG.col.anesthesiaEnd).clearContent();
      }
      continue;
    }

    const endDate = new Date(startDate.getTime());
    endDate.setMinutes(endDate.getMinutes() + minutes);

    const expectedEndDisplay = Utilities.formatDate(
      endDate,
      Session.getScriptTimeZone(),
      'HH:mm'
    );

    if (endDisplayValue !== expectedEndDisplay) {
      const endCell = sheet.getRange(currentRow, CONFIG.col.anesthesiaEnd);
      endCell.setValue(endDate);
      endCell.setNumberFormat('HH:mm');
    }
  }
}

function backfillAnesthesiaEndTime() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.mainSheetName);

  if (!sheet) return;

  const lastRow = sheet.getLastRow();
  if (lastRow < CONFIG.startRow) return;

  calculateAnesthesiaEndTime_(
    sheet,
    CONFIG.startRow,
    lastRow - CONFIG.startRow + 1,
    false
  );
}


/****************************************************
 * 고객별 굵은 구분선
 * - E열 고객명이 달라지는 첫 행에만 굵은 위쪽 선
 * - 기존에 잘못 남은 행별 위/아래 테두리를 완전히 제거
 ****************************************************/

/**
 * E열 고객명이 수정되면 수정된 부분 주변의 구분선만 갱신
 */
function drawCustomerGroupLines_(e) {
  if (!e || !e.range) return;

  const sheet = e.range.getSheet();
  if (sheet.getName() !== CONFIG.mainSheetName) return;

  const range = e.range;
  const customerCol = CONFIG.col.customer;

  if (
    range.getColumn() > customerCol ||
    range.getLastColumn() < customerCol
  ) {
    return;
  }

  const firstEditedRow = Math.max(range.getRow(), CONFIG.startRow);
  const lastEditedRow = range.getLastRow();
  const lastSheetRow = sheet.getLastRow();

  /*
   * 고객명 변경은 현재 행뿐 아니라
   * 바로 이전·다음 그룹의 시작 여부에도 영향을 줄 수 있음
   */
  const checkStartRow = Math.max(
    CONFIG.startRow,
    firstEditedRow - 1
  );

  const checkEndRow = Math.min(
    lastSheetRow,
    lastEditedRow + 1
  );

  updateCustomerGroupLinesNearby_(
    sheet,
    checkStartRow,
    checkEndRow
  );
}


/**
 * 지정된 행 범위의 구분선만 다시 판단
 */
function updateCustomerGroupLinesNearby_(sheet, startRow, endRow) {
  if (endRow < startRow) return;

  const customerCol = CONFIG.col.customer;
  const lastColumn = col_('AE');

  // 앞뒤 고객명을 확인하기 위해 한 행씩 추가로 읽음
  const readStartRow = Math.max(CONFIG.startRow, startRow - 1);
  const readEndRow = Math.min(sheet.getLastRow(), endRow + 1);
  const readRowCount = readEndRow - readStartRow + 1;

  const customerNames = sheet
    .getRange(readStartRow, customerCol, readRowCount, 1)
    .getDisplayValues()
    .map(function(row) {
      return normalizeCustomerNameForBorder_(row[0]);
    });

  function customerNameAt_(rowNumber) {
    if (rowNumber < readStartRow || rowNumber > readEndRow) {
      return '';
    }

    return customerNames[rowNumber - readStartRow];
  }

  for (let row = startRow; row <= endRow; row++) {
    const currentName = customerNameAt_(row);
    const previousName = customerNameAt_(row - 1);
    const nextName = customerNameAt_(row + 1);

    // 기존 코드와 동일한 조건:
    // 같은 고객이 2행 이상 이어지는 그룹의 첫 행
    const shouldHaveTopBorder =
      currentName !== '' &&
      currentName !== previousName &&
      currentName === nextName;

    const targetRange = sheet.getRange(
      row,
      1,
      1,
      lastColumn
    );

    targetRange.setBorder(
      shouldHaveTopBorder ? true : false,
      null,
      null,
      null,
      null,
      null,
      shouldHaveTopBorder ? '#000000' : null,
      shouldHaveTopBorder
        ? SpreadsheetApp.BorderStyle.SOLID_MEDIUM
        : null
    );
  }
}


function normalizeCustomerNameForBorder_(value) {
  return String(value || '')
    .replace(/[\u200B-\u200D\uFEFF]/g, '')
    .replace(/\s+/g, ' ')
    .trim()
    .toUpperCase();
}



/****************************************************
 * 대기 현황판
 ****************************************************/

function refreshWaitingDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.dashboardSheetName);

  if (!sheet) {
    console.warn('대기 현황판 시트를 찾을 수 없음: ' + CONFIG.dashboardSheetName);
    return;
  }

  const config = {
    dataStartRow: 2,

    receiptTimeCol: col_('S'),
    treatmentNameCol: col_('W'),
    quantityCol: col_('X'),
    hiddenCol: col_('Y'),

    doctorNameCol: col_('AB'),
    doctorFinishCol: col_('AC'),

    treatmentFinishCol: col_('AJ'),
    skin1Col: col_('AK'),

    doctorStartTimeCol: col_('AR'),
    skinStartTimeCol: col_('AS'),

    doctorDurationCol: col_('AT'),
    prepDurationCol: col_('AU'),
    finishDurationCol: col_('AV'),

    dashboardMachineCol: col_('L'),
    dashboardWriteCol: col_('M'),
    dashboardStartRow: 2,
    dashboardEndRow: 12,
    dashboardWriteCols: 3
  };

  const lastRow = sheet.getLastRow();
  const dashboardRows = config.dashboardEndRow - config.dashboardStartRow + 1;

  const machineNames = sheet
    .getRange(config.dashboardStartRow, config.dashboardMachineCol, dashboardRows, 1)
    .getDisplayValues()
    .map(function(row) {
      return String(row[0] || '').trim();
    });

  if (lastRow < config.dataStartRow) {
    const emptyOutput = machineNames.map(function(name) {
      return name ? ['0분', '0분', '0분'] : ['', '', ''];
    });

    writeDashboardValuesOnly_(sheet, config, emptyOutput);
    return;
  }

  const numRows = lastRow - config.dataStartRow + 1;
  const maxCol = col_('AV');

  const values = sheet.getRange(config.dataStartRow, 1, numRows, maxCol).getValues();
  const displayValues = sheet.getRange(config.dataStartRow, 1, numRows, maxCol).getDisplayValues();

  const now = new Date();
  const output = [];

  for (let m = 0; m < machineNames.length; m++) {
    const machineName = machineNames[m];

    if (!machineName) {
      output.push(['', '', '']);
      continue;
    }

    const machine = getMachineConfig_(machineName);

    let runningMinutes = 0;
    let waitingMinutes = 0;

    for (let i = 0; i < numRows; i++) {
      const row = values[i];
      const displayRow = displayValues[i];

      const receiptTime = row[config.receiptTimeCol - 1];
      const treatmentName = displayRow[config.treatmentNameCol - 1];
      const hidden = row[config.hiddenCol - 1];

      if (isBlank_(receiptTime)) continue;
      if (!matchesKeywords_(treatmentName, machine.keywords)) continue;
      if (isTrue_(hidden)) continue;

      const finishMinutes = parseMinutes_(
        row[config.finishDurationCol - 1],
        displayRow[config.finishDurationCol - 1]
      );

      if (machine.type === 'skin_quantity') {
        const treatmentFinish = row[config.treatmentFinishCol - 1];
        if (isTrue_(treatmentFinish)) continue;

        const quantityMinutes = parseMinutes_(
          row[config.quantityCol - 1],
          displayRow[config.quantityCol - 1]
        );

        const skin1 = row[config.skin1Col - 1];
        const skinStartTime = row[config.skinStartTimeCol - 1];
        const skinStartDisplay = displayRow[config.skinStartTimeCol - 1];

        const hasSkin1 = !isBlank_(skin1);
        const startDate = parseTimeToToday_(skinStartTime, skinStartDisplay, now);

        const totalMinutes = quantityMinutes + finishMinutes;

        if (!hasSkin1 || !startDate || startDate.getTime() > now.getTime()) {
          waitingMinutes += totalMinutes;
          continue;
        }

        const elapsed = diffMinutes_(startDate, now);
        runningMinutes += Math.max(totalMinutes - elapsed, 0);
        continue;
      }

      const doctorName = row[config.doctorNameCol - 1];
      const doctorFinish = row[config.doctorFinishCol - 1];
      const treatmentFinish = row[config.treatmentFinishCol - 1];

      if (isTrue_(doctorFinish) || isTrue_(treatmentFinish)) continue;

      const prepMinutes = parseMinutes_(
        row[config.prepDurationCol - 1],
        displayRow[config.prepDurationCol - 1]
      );

      const doctorDuration = parseMinutes_(
        row[config.doctorDurationCol - 1],
        displayRow[config.doctorDurationCol - 1]
      );

      const startTime = row[config.doctorStartTimeCol - 1];
      const startDisplay = displayRow[config.doctorStartTimeCol - 1];
      const startDate = parseTimeToToday_(startTime, startDisplay, now);

      const waitingTotal = prepMinutes + doctorDuration + finishMinutes;
      const runningTotal = doctorDuration + finishMinutes;

      if (isBlank_(doctorName) || !startDate || startDate.getTime() > now.getTime()) {
        waitingMinutes += waitingTotal;
        continue;
      }

      const elapsed = diffMinutes_(startDate, now);
      runningMinutes += Math.max(runningTotal - elapsed, 0);
    }

    const expectedMinutes = machine.machineCount > 0
      ? Math.ceil((runningMinutes + waitingMinutes) / machine.machineCount)
      : 0;

    output.push([
      minutesToWaitText_(runningMinutes),
      minutesToWaitText_(waitingMinutes),
      minutesToWaitText_(expectedMinutes)
    ]);
  }

  writeDashboardValuesOnly_(sheet, config, output);
}

function writeDashboardValuesOnly_(sheet, config, output) {
  const dashboardRows = config.dashboardEndRow - config.dashboardStartRow + 1;

  const range = sheet.getRange(
    config.dashboardStartRow,
    config.dashboardWriteCol,
    dashboardRows,
    config.dashboardWriteCols
  );

  range.clearContent();

  if (!output.length) return;

  const writeRows = Math.min(output.length, dashboardRows);

  sheet
    .getRange(config.dashboardStartRow, config.dashboardWriteCol, writeRows, config.dashboardWriteCols)
    .setValues(output.slice(0, writeRows))
    .setNumberFormat('@');
}

function col_(letter) {
  const text = String(letter || '').toUpperCase().replace(/[^A-Z]/g, '');
  let column = 0;

  for (let i = 0; i < text.length; i++) {
    column = column * 26 + text.charCodeAt(i) - 64;
  }

  return column;
}

function isBlank_(value) {
  if (value === null || value === undefined) return true;

  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    return false;
  }

  return String(value).trim() === '';
}

function isTrue_(value) {
  if (value === true) return true;

  const text = String(value || '').trim().toLowerCase();

  return (
    text === 'true' ||
    text === '1' ||
    text === 'y' ||
    text === 'yes' ||
    text === 'o' ||
    text === '완료' ||
    text === '체크'
  );
}

function getCurrentTimeOnlyValue_() {
  const now = new Date();

  const totalSeconds =
    now.getHours() * 60 * 60 +
    now.getMinutes() * 60 +
    now.getSeconds();

  return totalSeconds / (24 * 60 * 60);
}

function normalizeToTimeOnlyValue_(value) {
  if (isBlank_(value)) return '';

  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    const totalSeconds =
      value.getHours() * 60 * 60 +
      value.getMinutes() * 60 +
      value.getSeconds();

    return totalSeconds / (24 * 60 * 60);
  }

  if (typeof value === 'number' && isFinite(value)) {
    return value % 1;
  }

  const text = String(value).trim();
  if (!text) return '';

  const match = text.match(/(오전|오후|AM|PM|am|pm)?\s*(\d{1,2})\s*[:시]\s*(\d{1,2})?/);
  if (!match) return value;

  let hour = Number(match[2]);
  const minute = Number(match[3] || 0);
  const ampm = String(match[1] || '').toLowerCase();

  if ((ampm === '오후' || ampm === 'pm') && hour < 12) hour += 12;
  if ((ampm === '오전' || ampm === 'am') && hour === 12) hour = 0;

  if (!isFinite(hour) || !isFinite(minute)) return value;
  if (hour < 0 || hour > 23 || minute < 0 || minute > 59) return value;

  return (hour * 60 + minute) / (24 * 60);
}

function parseTimeValue_(value, displayValue) {
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) {
    return new Date(value);
  }

  if (typeof value === 'number' && isFinite(value)) {
    const date = new Date();
    const totalMinutes = Math.round((value % 1) * 24 * 60);

    date.setHours(Math.floor(totalMinutes / 60));
    date.setMinutes(totalMinutes % 60);
    date.setSeconds(0);
    date.setMilliseconds(0);

    return date;
  }

  const text = String(displayValue || value || '').trim();

  const match = text.match(/^(오전|오후|AM|PM|am|pm)?\s*(\d{1,2})[:시]\s*(\d{1,2})?/);
  if (!match) return null;

  const ampm = String(match[1] || '').toLowerCase();
  let hour = Number(match[2]);
  const minute = Number(match[3] || 0);

  if ((ampm === '오후' || ampm === 'pm') && hour < 12) hour += 12;
  if ((ampm === '오전' || ampm === 'am') && hour === 12) hour = 0;

  if (hour < 0 || hour > 23 || minute < 0 || minute > 59) return null;

  const date = new Date();
  date.setHours(hour);
  date.setMinutes(minute);
  date.setSeconds(0);
  date.setMilliseconds(0);

  return date;
}

function parseTimeToToday_(value, displayValue, baseDate) {
  const now = baseDate || new Date();
  const parsed = parseTimeValue_(value, displayValue);

  if (!parsed) return null;

  return new Date(
    now.getFullYear(),
    now.getMonth(),
    now.getDate(),
    parsed.getHours(),
    parsed.getMinutes(),
    parsed.getSeconds()
  );
}

function parseMinutes_(value, displayValue) {
  if (typeof value === 'number' && !isNaN(value)) {
    return Math.max(Math.ceil(value), 0);
  }

  const text = String(displayValue || value || '').trim();
  if (!text) return 0;

  let total = 0;

  const hourMatch = text.match(/(\d+(?:\.\d+)?)\s*시간/);
  if (hourMatch) total += Number(hourMatch[1]) * 60;

  const minuteMatch = text.match(/(\d+(?:\.\d+)?)\s*분/);
  if (minuteMatch) total += Number(minuteMatch[1]);

  if (total > 0) return Math.max(Math.ceil(total), 0);

  const numberOnly = text.match(/(\d+(?:\.\d+)?)/);
  if (numberOnly) return Math.max(Math.ceil(Number(numberOnly[1])), 0);

  return 0;
}

function diffMinutes_(startDate, endDate) {
  if (!startDate || !endDate) return 0;

  return Math.max(
    Math.floor((endDate.getTime() - startDate.getTime()) / 60000),
    0
  );
}

function minutesToWaitText_(minutes) {
  const safeMinutes = Math.max(Math.ceil(Number(minutes) || 0), 0);
  const hours = Math.floor(safeMinutes / 60);
  const mins = safeMinutes % 60;

  if (hours <= 0) return mins + '분';
  if (mins === 0) return hours + '시간';

  return hours + '시간 ' + mins + '분';
}

function normalizeText_(value) {
  return String(value || '')
    .toLowerCase()
    .replace(/\s+/g, '')
    .replace(/[()（）\[\]【】]/g, '')
    .trim();
}

function matchesKeywords_(text, keywords) {
  const normalizedText = normalizeText_(text);
  if (!normalizedText) return false;

  return keywords.some(function(keyword) {
    const normalizedKeyword = normalizeText_(keyword);
    return normalizedKeyword && normalizedText.indexOf(normalizedKeyword) !== -1;
  });
}

function extractMachineCount_(machineName) {
  const text = String(machineName || '');
  const match = text.match(/(\d+)\s*(대|개|ea|EA)?/);

  if (!match) return 1;

  const count = Number(match[1]);
  if (!count || count < 1) return 1;

  return count;
}
